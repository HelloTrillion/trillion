> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Trillion loads `SKILL.md` skills from a curated set and from folders the user adds. Each starts off, switches back off when its files change, and is read through two read-only tools. Scripts are never run. (CON-6, spec: `docs/specs/2026-10-23-skills.md`)

**Architecture:** A new `trillion.skills` package: `parse` reads one skill folder, `fingerprint` hashes it, `library` finds skills and keeps the user's choices in `~/.trillion/skills.json`. Two first-party tools, `skill_open` and `skill_file`, read only skills that are on and unchanged. The system block lists the enabled skills. A FastAPI router and a Settings section switch them; `trillion doctor` reports them.

**Tech Stack:** Python 3.13, PyYAML (`yaml.safe_load`), FastAPI, pytest; TypeScript, Vite, vitest.

```python
"""Reading one skill folder: the header, the instructions, and the files beside them.

The header follows the Agent Skills format: YAML between two `---` lines, with a
required `name` and `description`. Anything that breaks a rule or a limit makes the
skill invalid, with a reason the user can read. Nothing here runs or imports a file.
"""

from __future__ import annotations

import os
import re
import stat
from collections.abc import Iterator
from dataclasses import dataclass
from pathlib import Path

import yaml

SKILL_MD = "SKILL.md"
MAX_SKILL_MD = 64 * 1024
MAX_FILE = 256 * 1024
MAX_FILES = 200
MAX_DEPTH = 8
MAX_NAME = 64
MAX_DESCRIPTION = 1024
_NAME = re.compile(r"^[a-z0-9]+(-[a-z0-9]+)*$")
SCRIPT_SUFFIXES = frozenset({".py", ".sh", ".bash", ".zsh", ".js", ".mjs", ".ts", ".rb", ".ps1"})
# What editors and operating systems leave behind: never part of a skill.
IGNORED = frozenset({".DS_Store", "Thumbs.db", "desktop.ini", "__pycache__", ".git"})


@dataclass(frozen=True, kw_only=True)
class SkillFile:
    path: str  # relative to the skill's folder, with forward slashes
    size: int
    script: bool
    text: bool


@dataclass(frozen=True, kw_only=True)
class Skill:
    name: str
    description: str
    instructions: str
    allowed_tools: tuple[str, ...]
    license: str
    files: tuple[SkillFile, ...]  # every file but SKILL.md

    @property
    def has_scripts(self) -> bool:
        return any(f.script for f in self.files)


@dataclass(frozen=True)
class Invalid:
    reason: str


def walk(folder: Path) -> Iterator[tuple[str, Path, int]]:
    """Every file under `folder` as (relative path, path, depth), symlinks included and
    never followed, ignored names skipped, in sorted order."""
    for root, dirs, files in os.walk(folder, followlinks=False):
        here = Path(root)
        depth = len(here.relative_to(folder).parts)
        dirs[:] = sorted(d for d in dirs if d not in IGNORED and not (here / d).is_symlink())
        links = [d for d in os.listdir(here) if (here / d).is_symlink() and (here / d).is_dir()]
        for name in sorted({*files, *links}):
            if name in IGNORED:
                continue
            path = here / name
            yield path.relative_to(folder).as_posix(), path, depth


def _header(text: str) -> tuple[dict[str, object], str] | Invalid:
    text = text.removeprefix("\ufeff").replace("\r\n", "\n")
    if not text.startswith("---\n"):
        return Invalid("its SKILL.md has no header (the --- block at the top)")
    end = text.find("\n---", 4)
    if end < 0:
        return Invalid("its SKILL.md has no header (the --- block at the top)")
    try:
        data = yaml.safe_load(text[4:end])
    except yaml.YAMLError:
        return Invalid("its header could not be read")
    if not isinstance(data, dict):
        return Invalid("its header is not a set of fields")
    body = text[end + 4 :].removeprefix("\n")
    return data, body.strip()


def _tools(raw: object) -> tuple[str, ...]:
    if isinstance(raw, str):
        return tuple(t for t in re.split(r"[\s,]+", raw) if t)
    if isinstance(raw, list):
        return tuple(str(t).strip() for t in raw if str(t).strip())
    return ()


def _is_text(path: Path) -> bool:
    try:
        path.read_bytes().decode("utf-8")
    except (OSError, UnicodeDecodeError):
        return False
    return True


def _is_script(path: Path, mode: int) -> bool:
    return path.suffix.lower() in SCRIPT_SUFFIXES or (
        stat.S_ISREG(mode) and bool(mode & 0o111) and os.name != "nt"
    )


def parse(folder: Path) -> Skill | Invalid:
    md = folder / SKILL_MD
    try:
        size = md.lstat().st_size
    except OSError:
        return Invalid("it has no SKILL.md")
    if md.is_symlink() or not md.is_file():
        return Invalid("its SKILL.md is not a plain file")
    if size > MAX_SKILL_MD:
        return Invalid(f"its SKILL.md is too large (over {MAX_SKILL_MD // 1024} KB)")
    try:
        text = md.read_text(encoding="utf-8")
    except (OSError, UnicodeDecodeError):
        return Invalid("its SKILL.md is not readable text")
    header = _header(text)
    if isinstance(header, Invalid):
        return header
    fields, instructions = header
    name = str(fields.get("name") or "").strip()
    description = " ".join(str(fields.get("description") or "").split())
    if not name:
        return Invalid("its header has no name")
    if len(name) > MAX_NAME or not _NAME.match(name):
        return Invalid(
            "its name must be lowercase letters, digits and single hyphens, "
            f"up to {MAX_NAME} characters"
        )
    if not description:
        return Invalid("its header has no description")
    if len(description) > MAX_DESCRIPTION:
        return Invalid(f"its description is longer than {MAX_DESCRIPTION} characters")
    found: list[SkillFile] = []
    for rel, path, depth in walk(folder):
        if depth >= MAX_DEPTH:
            return Invalid(f"its folders are too deep (over {MAX_DEPTH} levels)")
        if rel == SKILL_MD:
            continue
        if len(found) >= MAX_FILES:
            return Invalid(f"it has too many files (over {MAX_FILES})")
        info = path.lstat()
        if info.st_size > MAX_FILE:
            return Invalid(f"{rel} is too large (over {MAX_FILE // 1024} KB)")
        link = stat.S_ISLNK(info.st_mode)
        found.append(
            SkillFile(
                path=rel,
                size=info.st_size,
                script=not link and _is_script(path, info.st_mode),
                text=not link and _is_text(path),
            )
        )
    return Skill(
        name=name,
        description=description,
        instructions=instructions,
        allowed_tools=_tools(fields.get("allowed-tools")),
        license=str(fields.get("license") or "").strip(),
        files=tuple(found),
    )
```

```python
"""One hash over a skill's files: what the user approved when they turned it on."""

from __future__ import annotations

import hashlib
import os
from pathlib import Path

from trillion.skills.parse import walk


def fingerprint(folder: Path) -> str:
    """SHA-256 over every file's relative path and bytes, in sorted order. A symlink
    is hashed by where it points, never followed, so retargeting one changes it."""
    digest = hashlib.sha256()
    for rel, path, _ in walk(folder):
        digest.update(rel.encode() + b"\0")
        if path.is_symlink():
            digest.update(b"link\0" + os.readlink(path).encode())
        else:
            try:
                digest.update(b"file\0" + path.read_bytes())
            except OSError:
                digest.update(b"unreadable\0")
        digest.update(b"\0")
    return "sha256:" + digest.hexdigest()
```

```python
"""Where skills come from, which are on, and whether they changed since (CON-6).

Skills come from the curated set shipped in this package and from folders the user
added. A folder is a skill if it has a SKILL.md; otherwise each direct subfolder that
has one is. The user's choices are in skills.json: the folders, and for each skill
turned on, the fingerprint they approved. A skill whose files no longer match is
"changed", and counts as off until the user turns it on again.
"""

from __future__ import annotations

import contextlib
import json
import os
import tempfile
from dataclasses import dataclass
from pathlib import Path
from typing import Any, Literal

from trillion import paths
from trillion.skills.fingerprint import fingerprint
from trillion.skills.parse import SKILL_MD, Invalid, Skill, parse

CURATED = Path(__file__).resolve().parent / "curated"
State = Literal["off", "on", "changed", "invalid"]
Source = Literal["curated", "folder"]


class LibraryError(ValueError):
    """Something the user asked for cannot be done. The message is safe to show."""


@dataclass(frozen=True, kw_only=True)
class Entry:
    id: str
    source: Source
    folder: Path  # the skill's own folder
    root: Path | None  # the user's folder it was found in; None when curated
    skill: Skill | None
    reason: str  # why it is invalid, else ""
    fingerprint: str
    state: State

    @property
    def name(self) -> str:
        return self.skill.name if self.skill else self.folder.name


@dataclass(frozen=True)
class Folder:
    path: Path
    found: bool


class Library:
    def __init__(
        self, settings_path: Path, curated_root: Path = CURATED, *, data_root: Path | None = None
    ) -> None:
        self._path = settings_path
        self._curated = curated_root
        self._data = (data_root or paths.home()).expanduser().resolve()

    # ── The settings file ─────────────────────────────────────────────

    def _read(self, *, strict: bool) -> dict[str, Any]:
        empty: dict[str, Any] = {"folders": [], "enabled": {}}
        try:
            raw = json.loads(self._path.read_text(encoding="utf-8"))
        except FileNotFoundError:
            return empty
        except (OSError, ValueError):
            if strict:
                raise LibraryError(
                    "skills.json could not be read; fix or remove it first"
                ) from None
            return empty
        if not isinstance(raw, dict):
            if strict:
                raise LibraryError("skills.json is not in the right shape; fix or remove it")
            return empty
        folders = [str(f) for f in raw.get("folders", []) if isinstance(f, str)]
        enabled = {
            str(k): str(v) for k, v in dict(raw.get("enabled") or {}).items() if isinstance(v, str)
        }
        return {"folders": folders, "enabled": enabled}

    def _write(self, data: dict[str, Any]) -> None:
        self._path.parent.mkdir(parents=True, exist_ok=True)
        fd, tmp = tempfile.mkstemp(prefix=".skills-", suffix=".json", dir=self._path.parent)
        try:
            with os.fdopen(fd, "w", encoding="utf-8") as f:
                json.dump(data, f, indent=2, ensure_ascii=False)
                f.write("\n")
            os.chmod(tmp, 0o600)
            os.replace(tmp, self._path)
        except BaseException:
            with contextlib.suppress(OSError):
                os.unlink(tmp)
            raise

    # ── Finding skills ────────────────────────────────────────────────

    def folders(self) -> list[Folder]:
        return [Folder(Path(f), Path(f).is_dir()) for f in self._read(strict=False)["folders"]]

    def _candidates(self) -> list[tuple[str, Source, Path, Path | None]]:
        out: list[tuple[str, Source, Path, Path | None]] = []
        if self._curated.is_dir():
            for sub in sorted(self._curated.iterdir()):
                if (sub / SKILL_MD).is_file():
                    out.append((f"curated:{sub.name}", "curated", sub, None))
        for f in self.folders():
            if not f.found:
                continue
            subs = [f.path] if (f.path / SKILL_MD).exists() else sorted(f.path.iterdir())
            for sub in subs:
                if sub.is_dir() and not sub.is_symlink() and (sub / SKILL_MD).exists():
                    out.append((f"folder:{sub.resolve()}", "folder", sub.resolve(), f.path))
        return out

    def _entry(
        self, sid: str, source: Source, folder: Path, root: Path | None, approved: str | None
    ) -> Entry:
        found = parse(folder)
        mark = fingerprint(folder)
        skill = found if isinstance(found, Skill) else None
        state: State
        if isinstance(found, Invalid):
            state = "invalid"
        elif approved is None:
            state = "off"
        else:
            state = "on" if approved == mark else "changed"
        return Entry(
            id=sid,
            source=source,
            folder=folder,
            root=root,
            skill=skill,
            reason=found.reason if isinstance(found, Invalid) else "",
            fingerprint=mark,
            state=state,
        )

    def scan(self) -> list[Entry]:
        enabled = self._read(strict=False)["enabled"]
        return [self._entry(*c, enabled.get(c[0])) for c in self._candidates()]

    def enabled(self) -> list[Entry]:
        """The skills that are on and unchanged, by name. Only those are fingerprinted."""
        on = self._read(strict=False)["enabled"]
        if not on:
            return []
        out = [self._entry(*c, on[c[0]]) for c in self._candidates() if c[0] in on]
        return sorted((e for e in out if e.state == "on"), key=lambda e: (e.name, e.id))

    def lookup(self, ref: str) -> Entry:
        """A skill that is on, by id or by a name only one skill that is on has. A skill
        whose files changed is returned with state "changed", for the caller to refuse."""
        ref = ref.strip()
        on = self._read(strict=False)["enabled"]
        found = [
            self._entry(*c, on[c[0]])
            for c in self._candidates()
            if c[0] in on and (c[0] == ref or self._named(c[2], ref))
        ]
        if not found:
            raise LibraryError(f"no skill called {ref!r} is turned on")
        if len(found) > 1:
            raise LibraryError(
                f"more than one skill is called {ref!r}; use its id: "
                + ", ".join(e.id for e in found)
            )
        return found[0]

    @staticmethod
    def _named(folder: Path, ref: str) -> bool:
        found = parse(folder)
        return isinstance(found, Skill) and found.name == ref

    def prompt_lines(self) -> list[tuple[str, str]]:
        """(label, description) for each skill that is on. A name two skills share is
        labelled with the folder each came from."""
        on = self.enabled()
        names = [e.name for e in on]
        out: list[tuple[str, str]] = []
        for e in on:
            label = e.name
            if names.count(e.name) > 1 and e.root is not None:
                label = f"{e.name} (from {e.root})"
            out.append((label, e.skill.description if e.skill else ""))
        return out

    # ── Changing the user's choices ───────────────────────────────────

    def turn_on(self, sid: str) -> Entry:
        data = self._read(strict=True)
        match = [c for c in self._candidates() if c[0] == sid]
        if not match:
            raise LibraryError("there is no such skill")
        entry = self._entry(*match[0], None)
        if entry.state == "invalid":
            raise LibraryError(f"that skill can't be used: {entry.reason}")
        data["enabled"][sid] = entry.fingerprint
        self._write(data)
        return self._entry(*match[0], entry.fingerprint)

    def turn_off(self, sid: str) -> None:
        data = self._read(strict=True)
        if data["enabled"].pop(sid, None) is not None:
            self._write(data)

    def add_folder(self, raw: str) -> Path:
        path = Path(raw.strip()).expanduser()
        if not path.is_dir():
            raise LibraryError("that is not a folder on this computer")
        path = path.resolve()
        if path == Path(path.anchor):
            raise LibraryError("a whole drive is too much; choose the folder your skills are in")
        if path == self._data or path.is_relative_to(self._data) or self._data.is_relative_to(path):
            raise LibraryError("that holds Trillion's own files; choose another folder")
        data = self._read(strict=True)
        if str(path) not in data["folders"]:
            data["folders"].append(str(path))
            self._write(data)
        return path

    def remove_folder(self, raw: str) -> None:
        data = self._read(strict=True)
        path = str(Path(raw.strip()).expanduser().resolve())
        data["folders"] = [f for f in data["folders"] if f != path]
        prefix = f"folder:{path}"
        data["enabled"] = {
            k: v
            for k, v in data["enabled"].items()
            if not (k == prefix or k.startswith(prefix + os.sep))
        }
        self._write(data)

    def fact(self) -> str:
        """For describe_self (BRN-6)."""
        on = self.enabled()
        if not on:
            return "none turned on"
        return (
            f"{len(on)} turned on: {', '.join(e.name for e in on)}. skill_open reads one; "
            "skill_file reads a file in it. Their scripts are never run"
        )
```

```python
"""skill_open and skill_file (CON-6): reading a skill the user turned on.

Both only read, so they run at once. What they return is text the user approved
byte for byte when they turned the skill on, so it is not wrapped as untrusted; it
still cannot change a gate, because the gates are code. Every call checks the
fingerprint again: a skill whose files changed is refused until the user looks at it.
"""

from __future__ import annotations

import re
from pathlib import PurePosixPath

from trillion.skills.library import Entry, Library, LibraryError
from trillion.skills.parse import SKILL_MD
from trillion.spine.contract import (
    JSON,
    BaseTool,
    Risk,
    ToolContext,
    ToolDefinition,
    ToolResult,
)

OPEN, FILE = "skill_open", "skill_file"
_DRIVE = re.compile(r"^[A-Za-z]:")
_SKILL_ARG = {
    "type": "string",
    "minLength": 1,
    "maxLength": 600,
    "description": "The skill's name as listed, or its id.",
}


def _refused(text: str, kind: str = "refused") -> ToolResult:
    return ToolResult(ok=False, content=text, effect="none", error_kind=kind)


def _kb(size: int) -> str:
    return f"{max(1, round(size / 1024))} KB"


class _SkillTool(BaseTool):
    def __init__(self) -> None:
        self._library: Library | None = None

    def bind(self, library: Library) -> None:
        """Called once at startup with the running process's library."""
        self._library = library

    def _entry(self, ref: object) -> Entry | ToolResult:
        if self._library is None:
            return _refused("skills are not available", "unavailable")
        try:
            entry = self._library.lookup(str(ref))
        except LibraryError as e:
            return _refused(f"{e}. Only skills the user turned on in Settings → Skills can be read")
        if entry.state != "on" or entry.skill is None:
            return _refused(
                f"the skill {entry.name!r} changed since it was turned on, so it is off until "
                "the user looks at it again in Settings → Skills",
                "changed",
            )
        return entry


class SkillOpen(_SkillTool):
    definition = ToolDefinition(
        name=OPEN,
        description=(
            "Read the instructions of one of the skills listed in your instructions, when "
            "the user's request matches what it says it is for. Returns its instructions "
            "and the list of its other files."
        ),
        input_schema={
            "type": "object",
            "properties": {"skill": _SKILL_ARG},
            "required": ["skill"],
            "additionalProperties": False,
        },
        risk=Risk.READ,
        category="skills",
        timeout_s=10,
        trusted_output=True,
    )

    async def execute(self, args: JSON, ctx: ToolContext) -> ToolResult:
        entry = self._entry(args.get("skill", ""))
        if isinstance(entry, ToolResult):
            return entry
        assert entry.skill is not None
        lines = [f'<skill name="{entry.name}">', entry.skill.instructions, "</skill>"]
        listed = [f for f in entry.skill.files if f.text]
        if listed:
            lines.append("Files in this skill (read one with skill_file):")
            for f in listed:
                note = ", a script: Trillion never runs it" if f.script else ""
                lines.append(f"- {f.path} ({_kb(f.size)}{note})")
        return ToolResult(ok=True, content="\n".join(lines), effect="none")


class SkillFile(_SkillTool):
    definition = ToolDefinition(
        name=FILE,
        description=(
            "Read one file from a skill you opened with skill_open, by its path as listed "
            "there. Scripts come back as text: you can read them, never run them."
        ),
        input_schema={
            "type": "object",
            "properties": {
                "skill": _SKILL_ARG,
                "path": {
                    "type": "string",
                    "minLength": 1,
                    "maxLength": 400,
                    "description": "The file's path inside the skill, as skill_open listed it.",
                },
            },
            "required": ["skill", "path"],
            "additionalProperties": False,
        },
        risk=Risk.READ,
        category="skills",
        timeout_s=10,
        # Inside the skill's own folder, never the granted folders: checked here.
        outside_path_args=("path",),
        trusted_output=True,
    )

    async def execute(self, args: JSON, ctx: ToolContext) -> ToolResult:
        entry = self._entry(args.get("skill", ""))
        if isinstance(entry, ToolResult):
            return entry
        assert entry.skill is not None
        raw = str(args.get("path", "")).strip()
        rel = PurePosixPath(raw.replace("\\", "/"))
        if not raw or rel.is_absolute() or _DRIVE.match(raw) or ".." in rel.parts:
            return _refused("give the file's path inside the skill, as skill_open listed it")
        name = rel.as_posix()
        if name == SKILL_MD:
            return _refused("use skill_open to read SKILL.md")
        listed = {f.path: f for f in entry.skill.files}
        found = listed.get(name)
        if found is None:
            return _refused("that file is not in this skill", "not_found")
        here = entry.folder
        for part in rel.parts:
            here = here / part
            if here.is_symlink():
                return _refused("that file is a link; links in skills are never followed")
        if not found.text or not here.resolve().is_relative_to(entry.folder.resolve()):
            return _refused("that file is not text Trillion can read")
        try:
            text = here.read_text(encoding="utf-8")
        except (OSError, UnicodeDecodeError):
            return _refused("that file could not be read", "unreadable")
        if found.script:
            text = "(A script. Trillion never runs it; read it to follow the skill.)\n" + text
        return ToolResult(ok=True, content=text, effect="none")
```

```python
"""Settings → Skills (CON-6). Behind the session token like every /api/ route.

Reads and changes skills.json through the library. Never returns a skill's
instructions: the page shows what the skill says it is for, not how.
"""

from __future__ import annotations

from typing import Any

from fastapi import APIRouter
from fastapi.responses import JSONResponse
from pydantic import BaseModel, Field

from trillion.api.runtime import Runtime
from trillion.skills.library import Entry, Library, LibraryError


class FolderBody(BaseModel):
    path: str = Field(min_length=1, max_length=1000)


class Switch(BaseModel):
    id: str = Field(min_length=1, max_length=1100)
    on: bool


def _fail(detail: str) -> JSONResponse:
    return JSONResponse({"error": "refused", "detail": detail}, status_code=400)


def create_skills_router(rt: Runtime) -> APIRouter:
    router = APIRouter()

    def library() -> Library:
        if rt.skills is None:
            raise RuntimeError("skills are not running")
        return rt.skills

    def row(e: Entry, have: set[str]) -> dict[str, Any]:
        s = e.skill
        return {
            "id": e.id,
            "source": e.source,
            "folder": str(e.root or ""),
            "name": e.name,
            "description": s.description if s else "",
            "license": s.license if s else "",
            "tools": [{"name": t, "available": t in have} for t in (s.allowed_tools if s else ())],
            "has_scripts": bool(s and s.has_scripts),
            "state": e.state,
            "reason": e.reason,
        }

    def state() -> dict[str, Any]:
        lib = library()
        have = {d.name for d in rt.registry.definitions()}
        return {
            "folders": [{"path": str(f.path), "found": f.found} for f in lib.folders()],
            "skills": [row(e, have) for e in lib.scan()],
        }

    @router.get("/api/skills")
    async def read() -> dict[str, Any]:
        return state()

    @router.post("/api/skills/folders")
    async def add(body: FolderBody) -> Any:
        try:
            library().add_folder(body.path)
        except LibraryError as e:
            return _fail(str(e))
        return state()

    @router.delete("/api/skills/folders")
    async def remove(body: FolderBody) -> Any:
        try:
            library().remove_folder(body.path)
        except LibraryError as e:
            return _fail(str(e))
        return state()

    @router.patch("/api/skills")
    async def switch(body: Switch) -> Any:
        try:
            if body.on:
                library().turn_on(body.id)
            else:
                library().turn_off(body.id)
        except LibraryError as e:
            return _fail(str(e))
        return state()

    return router
```

```python
def check_skills(path: Path) -> Check:
    """Skills (CON-6): how many are on, and any that changed, broke or went missing."""
    from trillion.skills.library import Library

    lib = Library(path)
    entries = lib.scan()
    on = [e for e in entries if e.state == "on"]
    changed = [e.name for e in entries if e.state == "changed"]
    broken = [e.name for e in entries if e.state == "invalid"]
    missing = [str(f.path) for f in lib.folders() if not f.found]
    problems = [
        *([f"changed: {', '.join(changed)}"] if changed else []),
        *([f"can't be used: {', '.join(broken)}"] if broken else []),
        *([f"folder not found: {', '.join(missing)}"] if missing else []),
    ]
    if problems:
        return Check(
            name="Skills",
            status="warn",
            detail=f"{len(on)} on; " + "; ".join(problems),
            fix="Open Settings → Skills to look at them again.",
        )
    if not on:
        return Check(name="Skills", status="info", detail="none turned on")
    return Check(name="Skills", status="ok", detail=f"{len(on)} on")
```

```ts
/**
 * Settings → Skills (CON-6): the curated set and the user's folders, each skill a card
 * with a switch. Every skill starts off; one whose files changed is off again until
 * the user looks and turns it back on. Scripts are named, and never run.
 */

import type { Host } from './panel';
import type { Api } from './settings';

export interface SkillRow {
  readonly id: string;
  readonly source: 'curated' | 'folder';
  readonly folder: string;
  readonly name: string;
  readonly description: string;
  readonly license: string;
  readonly tools: readonly { readonly name: string; readonly available: boolean }[];
  readonly has_scripts: boolean;
  readonly state: 'off' | 'on' | 'changed' | 'invalid';
  readonly reason: string;
}

export interface SkillsState {
  readonly folders: readonly { readonly path: string; readonly found: boolean }[];
  readonly skills: readonly SkillRow[];
}

export function skillState(s: Pick<SkillRow, 'state' | 'reason'>): string {
  if (s.state === 'on') return 'On';
  if (s.state === 'changed') return 'Changed since you turned it on';
  if (s.state === 'invalid') return `Can't be used: ${s.reason}`;
  return 'Off';
}

export function skillsSummary(s: SkillsState | null): string {
  const on = s?.skills.filter((k) => k.state === 'on').length ?? 0;
  const changed = s?.skills.filter((k) => k.state === 'changed').length ?? 0;
  if (!on && !changed) return 'None on';
  return changed ? `${on} on, ${changed} changed` : `${on} on`;
}

function el<K extends keyof HTMLElementTagNameMap>(tag: K, cls = '', text?: string): HTMLElementTagNameMap[K] {
  const n = document.createElement(tag);
  if (cls) n.className = cls;
  if (text !== undefined) n.textContent = text;
  return n;
}

export function createSkills(host: Host, api: Api, changed: () => void): { open(): Promise<void>; reveal(id: string): void; summary(): string } {
  let last: SkillsState | null = null;
  let error = '';
  let focus = '';

  const form = el('form', 'settings connected skills');
  form.addEventListener('submit', (ev) => ev.preventDefault());
  host.setAttribute('aria-labelledby', 'skills-title');
  host.replaceChildren(form);

  async function call(method: 'GET' | 'POST' | 'PATCH' | 'DELETE', path: string, body?: unknown): Promise<void> {
    try {
      const r = await api(method, path, body);
      const data = (await r.json().catch(() => ({}))) as Partial<SkillsState> & { detail?: string };
      if (r.ok && data.skills) {
        last = data as SkillsState;
        error = '';
      } else {
        error = data.detail ?? 'That did not work.';
      }
    } catch {
      error = 'Trillion could not be reached.';
    }
    changed();
    draw();
  }

  function card(s: SkillRow): HTMLElement {
    const box = el('div', `connected-card skills-card is-${s.state}`);
    box.id = `skill-${s.id}`;
    const head = el('div', 'connected-card-head');
    const words = el('span', 'connected-card-words');
    words.append(el('strong', '', s.name), el('small', '', s.description));
    const sw = el('label', 'connected-switch');
    const input = el('input');
    input.type = 'checkbox';
    input.setAttribute('role', 'switch');
    input.checked = s.state === 'on';
    input.disabled = s.state === 'invalid';
    input.setAttribute('aria-label', `${s.name} turned on`);
    input.addEventListener('change', () => void call('PATCH', '/api/skills', { id: s.id, on: input.checked }));
    sw.append(el('small', '', skillState(s)), input, el('span', 'connected-switch-track'));
    head.append(words, sw);
    box.append(head);
    const facts = el('ul', 'skills-facts');
    facts.append(el('li', '', s.source === 'curated' ? 'Included with Trillion' : `From ${s.folder}`));
    if (s.tools.length) {
      const li = el('li', '', 'Mentions: ');
      li.append(s.tools.map((t) => (t.available ? t.name : `${t.name} (not available in Trillion)`)).join(', '));
      facts.append(li);
    }
    if (s.has_scripts) facts.append(el('li', 'skills-scripts', "Includes scripts Trillion won't run"));
    if (s.license) facts.append(el('li', '', `License: ${s.license}`));
    if (s.state === 'changed') facts.append(el('li', 'skills-changed', 'Its files changed. Read it again, then turn it back on to use it.'));
    box.append(facts);
    return box;
  }

  function draw(): void {
    const title = el('h2', '', 'Skills');
    title.id = 'skills-title';
    const note = el('p', 'connected-note', 'Instructions she reads when a task calls for one. Each starts off: turn on the ones you trust. A skill can tell her how, never let her skip asking you.');
    const parts: HTMLElement[] = [title, note];
    if (error) parts.push(el('p', 'connected-error', error));
    const skills = last?.skills ?? [];
    const curated = skills.filter((s) => s.source === 'curated');
    const mine = skills.filter((s) => s.source === 'folder');
    parts.push(el('h3', '', 'Included with Trillion'), ...curated.map(card));
    parts.push(el('h3', '', 'Your folders'));
    for (const f of last?.folders ?? []) {
      const row = el('div', 'skills-folder');
      row.append(el('span', '', f.path + (f.found ? '' : ' (not found)')));
      const remove = el('button', 'connected-link', 'Remove');
      remove.type = 'button';
      remove.addEventListener('click', () => void call('DELETE', '/api/skills/folders', { path: f.path }));
      row.append(remove);
      parts.push(row);
    }
    parts.push(...mine.map(card));
    const add = el('div', 'skills-add');
    const input = el('input');
    input.type = 'text';
    input.placeholder = '~/.claude/skills';
    input.setAttribute('aria-label', 'A folder of skills');
    const go = el('button', 'connected-primary', 'Add folder');
    go.type = 'button';
    go.addEventListener('click', () => {
      if (input.value.trim()) void call('POST', '/api/skills/folders', { path: input.value.trim() });
    });
    add.append(input, go);
    parts.push(add);
    form.replaceChildren(...parts);
    if (focus) {
      const target = document.getElementById(`skill-${focus}`);
      target?.scrollIntoView({ block: 'center' });
      target?.classList.add('is-open');
      focus = '';
    }
  }

  return {
    async open() {
      await call('GET', '/api/skills');
    },
    reveal(id: string) {
      focus = id;
    },
    summary() {
      return skillsSummary(last);
    },
  };
}
```

SECURITY.md, a "Skills" section:

```markdown
## Skills

A skill is instructions, and instructions steer the model, so every skill starts off and the user turns it on after seeing what it says it does. What they approve is a fingerprint of every file in the skill; any change turns it off again. Skill text is not marked untrusted, because the user approved it byte for byte, but it cannot change a gate: the gates are code. Trillion has no tool that runs code, so a skill's scripts are only ever read. `skill_file` reads inside the skill's own folder and refuses `..`, absolute paths and symlinks.
```
