# API usage pill (BRN-4) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Read the existing `usage` table back as a month-to-date total, show it as a pill in the header, and list it per provider and model in Settings.

**Architecture:** A read-only store query (`usage_since`) feeds a new `GET /api/usage` router that works out the month start in the user's timezone. The web page fetches it on connect and after every turn, formats it with pure functions in `apps/web/src/usage.ts`, and shows a header pill that opens a new Settings section.

**Tech Stack:** Python 3 / FastAPI / aiosqlite, pytest; TypeScript / Vite / vitest.

**Spec:** `docs/specs/2026-10-22-api-usage.md`

## Global Constraints

- A NULL `cost_usd` is never counted as $0. Totals sum priced rows only; unpriced tokens are shown beside them.
- "This month" starts at midnight on the 1st in the profile timezone (`rt.profile.timezone`), else the machine's local time.
- No schema change, no new recording. Voice providers, per-agent split, price entry and caps are out of scope.
- `GET /api/usage` goes through the existing token + origin check in `app.py` (no new auth code).
- If the fetch fails, the pill hides; it never shows a stale or guessed number.
- Do not touch `src/trillion/spine/**` or `src/trillion/persona/**` (CODEOWNERS).
- Every commit: `git commit -s` (DCO). No push, PR or issue comment without the maintainer's go-ahead.
- Local loop before calling it done: `uv run ruff check . && uv run ruff format --check . && uv run mypy && uv run pytest`, then `pnpm typecheck && pnpm test && pnpm -r --if-present run build`.

```python
    async def usage_since(self, owner_id: str, since_ms: int) -> list[dict[str, Any]]:
        """Usage per provider and model since a moment (BRN-4). A NULL cost is unknown,
        not free: it is left out of `cost_usd` and counted in `unpriced_*` instead."""
        async with self._conn.execute(
            "SELECT provider, model, SUM(input_tokens), SUM(output_tokens),"
            " SUM(cache_read_tokens), SUM(cache_write_tokens), SUM(cost_usd),"
            " SUM(cost_usd IS NULL),"
            " SUM(CASE WHEN cost_usd IS NULL THEN input_tokens + output_tokens"
            " + cache_read_tokens + cache_write_tokens ELSE 0 END)"
            " FROM usage WHERE owner_id = ? AND created_at >= ?"
            " GROUP BY provider, model"
            " ORDER BY SUM(cost_usd) IS NULL, SUM(cost_usd) DESC,"
            " SUM(input_tokens + output_tokens + cache_read_tokens + cache_write_tokens) DESC",
            (owner_id, since_ms),
        ) as cur:
            names = (
                "provider", "model", "input_tokens", "output_tokens", "cache_read_tokens",
                "cache_write_tokens", "cost_usd", "unpriced_turns", "unpriced_tokens",
            )
            rows = [dict(zip(names, row, strict=True)) for row in await cur.fetchall()]
        for r in rows:
            # Float sums drift (0.5 + 0.7 = 1.2000000000000002); a micro-dollar is plenty.
            if r["cost_usd"] is not None:
                r["cost_usd"] = round(r["cost_usd"], 6)
        return rows
```

(SQLite's `SUM` skips NULLs and returns NULL when every value is NULL, which is exactly "unknown stays unknown". Run `uv run ruff format` after pasting.)

"""Month-to-date API usage, read back for the pill (BRN-4).

Read only: every model turn and compaction already writes a `usage` row. A turn
whose provider has no prices keeps its cost NULL, and is counted here as unpriced
tokens, never as $0, so the bill can never look smaller than it is.
"""

from __future__ import annotations

from datetime import datetime
from typing import Any
from zoneinfo import ZoneInfo

from fastapi import APIRouter

from trillion.api.runtime import OWNER, Runtime


def month_start(now_ms: int, zone: str) -> tuple[str, int]:
    """The month `now_ms` falls in, and when it began: midnight on the 1st in `zone`,
    or in this machine's local time when no zone is set."""
    # A naive datetime is local time, and .timestamp() reads it back as local time.
    now = datetime.fromtimestamp(now_ms / 1000, ZoneInfo(zone) if zone else None)
    start = now.replace(day=1, hour=0, minute=0, second=0, microsecond=0)
    return f"{start:%Y-%m}", int(start.timestamp() * 1000)


def create_usage_router(rt: Runtime) -> APIRouter:
    router = APIRouter(prefix="/api/usage")

    @router.get("")
    async def read() -> dict[str, Any]:
        month, since = month_start(rt.clock(), rt.profile.timezone if rt.profile else "")
        rows = await rt.conversations.usage_since(OWNER, since)
        priced = [r["cost_usd"] for r in rows if r["cost_usd"] is not None]
        return {
            "month": month,
            "since_ms": since,
            "rows": rows,
            "total_cost_usd": round(sum(priced), 6) if priced else None,
            "unpriced_tokens": sum(r["unpriced_tokens"] for r in rows),
        }

    return router
```

/**
 * This month's API usage, as the core reports it (BRN-4).
 *
 * A model without prices has no cost, and that is shown as tokens, never as $0:
 * a bill must never look smaller than it is.
 */

import type { Api } from './settings';

export interface UsageRow {
  readonly provider: string;
  readonly model: string;
  readonly input_tokens: number;
  readonly output_tokens: number;
  readonly cache_read_tokens: number;
  readonly cache_write_tokens: number;
  readonly cost_usd: number | null;
  readonly unpriced_turns: number;
  readonly unpriced_tokens: number;
}

export interface Usage {
  readonly month: string;
  readonly since_ms: number;
  readonly rows: readonly UsageRow[];
  readonly total_cost_usd: number | null;
  readonly unpriced_tokens: number;
}

export function tokens(n: number): string {
  if (n < 1_000) return String(n);
  if (n < 999_500) return `${Math.round(n / 1_000)}k`;
  return `${(n / 1_000_000).toFixed(1).replace(/\.0$/, '')}M`;
}

export function dollars(n: number): string {
  return n > 0 && n < 0.005 ? '<$0.01' : `$${n.toFixed(2)}`;
}

function monthName(month: string): string {
  const [year, m] = month.split('-').map(Number);
  return new Date(Date.UTC(year ?? 1970, (m ?? 1) - 1, 1)).toLocaleString('en', { month: 'long', timeZone: 'UTC' });
}

export function pill(u: Usage): string {
  if (!u.rows.length) return 'No API use yet';
  if (u.total_cost_usd === null) return `${tokens(u.unpriced_tokens)} tokens`;
  if (!u.unpriced_tokens) return dollars(u.total_cost_usd);
  return `${dollars(u.total_cost_usd)} · ${tokens(u.unpriced_tokens)} unpriced`;
}

export function sentence(u: Usage): string {
  const when = monthName(u.month);
  if (!u.rows.length) return `No API use yet in ${when}.`;
  if (u.total_cost_usd === null) return `API usage in ${when}: ${tokens(u.unpriced_tokens)} tokens, from models without prices.`;
  if (!u.unpriced_tokens) return `API usage in ${when}: ${dollars(u.total_cost_usd)}.`;
  return `API usage in ${when}: ${dollars(u.total_cost_usd)}, plus ${tokens(u.unpriced_tokens)} tokens from models without prices.`;
}

export function breakdown(u: Usage): { name: string; detail: string }[] {
  return u.rows.map((r) => ({
    name: `${r.provider} · ${r.model}`,
    detail: `${r.cost_usd === null ? 'No price' : dollars(r.cost_usd)} · ${tokens(r.input_tokens + r.cache_read_tokens + r.cache_write_tokens)} in, ${tokens(r.output_tokens)} out`,
  }));
}

/** This month's usage, or null when it cannot be read: the pill hides rather than guess. */
export async function fetchUsage(api: Api): Promise<Usage | null> {
  try {
    const response = await api('GET', '/api/usage');
    return response.ok ? ((await response.json()) as Usage) : null;
  } catch {
    return null;
  }
}
```

export function createUsageSettings(host: Host, load: () => Promise<Usage | null>): { open(): Promise<void> } {
  const { finish } = section(
    host,
    'usage',
    'API usage',
    'What the model providers report for this month so far. A model without prices counts tokens only. Voice services are not counted yet.',
  );
  const list = el('ul', 'settings-list');
  const total = el('p', 'settings-hint');
  finish(list, total);
  return {
    async open() {
      const u = await load();
      const rows = u ? breakdown(u) : [];
      list.replaceChildren(
        ...(rows.length
          ? rows.map((r) => {
              const item = el('li', 'settings-path', r.name);
              item.append(el('br'), el('span', 'settings-hint', r.detail));
              return item;
            })
          : [el('li', 'settings-empty', u ? 'Nothing yet this month.' : 'Usage could not be read right now.')]),
      );
      total.textContent = u ? sentence(u) : '';
      host.showModal();
    },
  };
}
